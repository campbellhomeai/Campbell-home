// Campbell Concierge AI // Voice + Touch Kiosk for Friends & Family Rental const propertyInfo = { wifi: "Scan the QR code on the kiosk to connect to Wi Fi.", checkout: "Checkout is at 11 AM. Please turn off lights and lock the door when leaving.", parking: "Guest parking is available in the rear lot.", laundry: "Laundry facilities are located in the hallway.", housekeeping: "You can request towels, linens, cleaning, or supplies.", maintenance: "You can report property issues through the maintenance screen.", thermostat: "The thermostat is located on the main hallway wall.", trash: "Please place trash in the designated disposal area.", emergency: "For emergencies call 911. For property emergencies contact the host.", host: "You may contact the host directly through the Contact Host button." }; // ---------- TEXT TO SPEECH ---------- function speak(text) { const speech = new SpeechSynthesisUtterance(text); speech.lang = "en-US"; speech.rate = 1; speech.pitch = 1; speech.volume = 1; window.speechSynthesis.cancel(); window.speechSynthesis.speak(speech); } // ---------- VOICE COMMAND ROUTER ---------- function processCommand(command) { const query = command.toLowerCase(); if (query.includes("wifi") || query.includes("internet")) { speak(propertyInfo.wifi); } else if (query.includes("checkout")) { speak(propertyInfo.checkout); } else if (query.includes("parking")) { speak(propertyInfo.parking); } else if (query.includes("laundry")) { speak(propertyInfo.laundry); } else if ( query.includes("housekeeping") || query.includes("towels") || query.includes("linens") ) { speak(propertyInfo.housekeeping); openServiceModal("housekeeping"); } else if ( query.includes("maintenance") || query.includes("broken") || query.includes("repair") ) { speak(propertyInfo.maintenance); openServiceModal("maintenance"); } else if (query.includes("thermostat")) { speak(propertyInfo.thermostat); } else if (query.includes("trash")) { speak(propertyInfo.trash); } else if ( query.includes("emergency") ) { speak(propertyInfo.emergency); } else if ( query.includes("host") || query.includes("contact") ) { speak(propertyInfo.host); contactHost(); } else { speak( "I'm Campbell Concierge. I can help with Wi Fi, parking, housekeeping, maintenance, checkout, and local information." ); } } // ---------- SPEECH RECOGNITION ---------- const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; const recognition = new SpeechRecognition(); recognition.continuous = true; recognition.interimResults = false; recognition.lang = "en-US"; recognition.onresult = (event) => { const transcript = event.results[event.results.length - 1][0].transcript; console.log("Guest said:", transcript); processCommand(transcript); }; recognition.onerror = (event) => { console.error(event.error); }; // ---------- WAKE WORD ---------- function startCampbellConcierge() { recognition.start(); speak( "Hello. I am Campbell Concierge. How may I assist you today?" ); } // ---------- SERVICE ACTIONS ---------- function openServiceModal(type) { console.log(`${type} request initiated`); // Future: // Create ticket // Notify property manager // Log guest request } function contactHost() { console.log("Connecting guest to host"); // Future: // SMS host // Call host // Launch chat interface } // ---------- STARTUP ---------- window.onload = () => { startCampbellConcierge(); };
